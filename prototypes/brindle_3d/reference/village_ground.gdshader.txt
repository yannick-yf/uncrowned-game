shader_type spatial;
uniform vec4 grass_dark : source_color = vec4(0.19, 0.28, 0.12, 1.0);
uniform vec4 grass_light : source_color = vec4(0.33, 0.41, 0.18, 1.0);
uniform vec4 dirt_color : source_color = vec4(0.56, 0.40, 0.24, 1.0);
uniform vec4 sand_color : source_color = vec4(0.70, 0.60, 0.40, 1.0);
uniform float path_width : hint_range(0.5, 2.0) = 1.0;
uniform sampler2D ground_noise : filter_linear_mipmap, repeat_enable;
uniform bool use_painted_ground = false;
uniform sampler2D painted_earth : source_color, filter_linear_mipmap_anisotropic, repeat_enable;
uniform float painted_strength : hint_range(0.0, 1.0) = 0.70;
uniform float painted_scale : hint_range(0.1, 1.0) = 0.3333;
uniform float edge_softness : hint_range(0.1, 1.5) = 0.8;
// Sentiers de l'extension, renseignés par le matériau de la map.
uniform int extra_path_count : hint_range(0, 40) = 0;
uniform vec4 extra_paths[40];
uniform float extra_path_radius : hint_range(0.5, 2.0) = 1.05;
varying vec3 world;
float noise(vec2 p) {
	return texture(ground_noise, p * 0.021).r;
}
float segment(vec2 p, vec2 a, vec2 b) {
	vec2 ab = b-a;
	return length(p-a-ab*clamp(dot(p-a,ab)/dot(ab,ab),0.0,1.0));
}
void vertex() { world = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz; }
void fragment() {
	vec2 p = world.xz;
	vec2 warp = vec2(noise(p*0.7), noise(p*0.7+72.0))-0.5;
	vec2 q = p + warp * 0.75;
	float d = segment(q, vec2(-23,-2), vec2(-5,-0.8))-0.90;
	d = min(d, segment(q,vec2(-5,-0.8),vec2(0,0))-1.0);
	d = min(d, segment(q,vec2(0,0),vec2(4,6.8))-1.05);
	d = min(d, segment(q,vec2(4,6.8),vec2(8,7.4))-0.9);
	d = min(d, segment(q,vec2(1.5,3),vec2(-5,11))-0.75);
	d = min(d, segment(q,vec2(-5.5,-1.1),vec2(-5.5,-3.1))-0.65);
	d = min(d, segment(q,vec2(8,7.4),vec2(8,5.8))-0.65);
	for (int i = 0; i < extra_path_count; i++) {
		d = min(d, segment(q, extra_paths[i].xy, extra_paths[i].zw) - extra_path_radius);
	}
	float grain = noise(p*18.0);
	float patches = noise(p*0.65)*0.58+noise(p*2.9)*0.27+noise(p*8.0)*0.15;
	vec3 grass = mix(grass_dark.rgb, grass_light.rgb, smoothstep(0.22,0.80,patches));
	grass *= 0.94 + grain*0.12;
	// Frange irrégulière ; le centre et le tracé praticable du chemin sont conservés.
	float edge = d/path_width + (noise(p*6.5)-0.5)*0.82;
	float bare = 1.0-smoothstep(-0.12,edge_softness,edge);
	vec3 dirt = dirt_color.rgb * (0.77 + noise(p*4.0)*0.28 + grain*0.19);
	float speckles = step(0.80,noise(p*14.0)) * step(0.40,noise(p*2.0));
	dirt = mix(dirt,dirt_color.rgb*1.45,speckles*0.30);
	if (use_painted_ground) {
		vec3 earth = texture(painted_earth, p * painted_scale).rgb;
		dirt = mix(dirt, earth * vec3(1.03, 1.02, 0.98), painted_strength);
		// De rares plages de terre relient visuellement le sentier à la prairie.
		float worn = smoothstep(0.60, 0.77, noise(p * 4.0));
		bare = max(bare, worn * (1.0 - smoothstep(0.2, 1.4, edge)) * 0.28);
	}
	float beach = smoothstep(8.0,10.9,q.y+(noise(p*2.0)-0.5)*1.0);
	vec3 sand = sand_color.rgb * (0.92 + noise(p*10.0)*0.10 + grain*0.08);
	float damp = smoothstep(13.5,15.3,p.y-sin(p.x*0.33)*0.5);
	sand *= 1.0 - damp*0.31;
	ALBEDO = mix(mix(grass,dirt,bare),sand,beach);
	ROUGHNESS = 0.96;
}
